class TimedResult(unittest.TextTestResult):
    """Report aggregate test-method wall time per explicitly loaded module."""
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.module_seconds = {}

    def startTest(self, test):
        self._started_at = time.perf_counter()
        super().startTest(test)

    def stopTest(self, test):
        module = test.__class__.__module__
        self.module_seconds[module] = self.module_seconds.get(module, 0.0) + (time.perf_counter() - self._started_at)
        super().stopTest(test)


def audit_sources():
    imports = {}
    for rel in SOURCES:
        path = ROOT / rel
        tree = ast.parse(path.read_text(encoding='utf-8'), filename=rel)
        found = set()
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                found.update(alias.name.split('.')[0] for alias in node.names)
            elif isinstance(node, ast.ImportFrom):
                if node.level or node.module is None:
                    raise ValueError('unreviewed relative import in ' + rel)
                found.add(node.module.split('.')[0])
        unknown = found - ALLOWED_IMPORTS
        if unknown:
            raise ValueError('unreviewed imports in ' + rel + ': ' + ', '.join(sorted(unknown)))
        imports[rel] = sorted(found)
    return imports


def main():
    started = time.perf_counter()
    if len(sys.argv) != 1:
        print('FAIL: no command-line test override accepted', file=sys.stderr)
        return 2
    if not (sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode):
        print('FAIL: run with Python -I -S -B', file=sys.stderr)
        return 2
    try:
        imports = audit_sources()
        sys.path.insert(0, str(VENDOR))
        sys.path.insert(0, str(ROOT))
        random.seed(SEED)
        # Guard common accidental Python socket connects. This is NOT an OS
        # network sandbox and cannot stop all native or subprocess access.
        with mock.patch.object(socket.socket, 'connect', side_effect=RuntimeError('network disabled in synthetic suite')), \
             mock.patch.object(socket.socket, 'connect_ex', side_effect=RuntimeError('network disabled in synthetic suite')), \
             mock.patch.object(socket, 'create_connection', side_effect=RuntimeError('network disabled in synthetic suite')):
            suite = unittest.TestSuite()
            for name in TEST_MODULES:
                module = importlib.import_module(name)
                if Path(module.__file__).resolve() != (ROOT / (name + '.py')).resolve():
                    raise ValueError('unexpected test module origin: ' + name)
                suite.addTests(unittest.defaultTestLoader.loadTestsFromModule(module))
            for name, path in (('research.ai_offline', VENDOR / 'research' / 'ai_offline.py'),
                               ('strategies.execution', VENDOR / 'strategies' / 'execution.py')):
                module = sys.modules.get(name)
                if module is None or Path(module.__file__).resolve() != path.resolve():
                    raise ValueError('unexpected vendor origin: ' + name)
            count = suite.countTestCases()
            if count == 0:
                raise ValueError('no tests loaded')
            result = unittest.TextTestRunner(verbosity=2, resultclass=TimedResult).run(suite)
        summary = {'suite': 'lightweight_synthetic_development_only', 'seed': SEED,
                   'tests': result.testsRun, 'failures': len(result.failures), 'errors': len(result.errors),
                   'skipped': len(result.skipped), 'expected_failures': len(result.expectedFailures),
                   'unexpected_successes': len(result.unexpectedSuccesses),
                   'seconds': round(time.perf_counter() - started, 6),
                   'status': 'PASS' if (result.wasSuccessful() and result.testsRun == count
                                        and not result.skipped and not result.expectedFailures
                                        and not result.unexpectedSuccesses) else 'FAIL',
                   'test_module_seconds': {k: round(v, 6) for k, v in sorted(result.module_seconds.items())},
                   'formal_22': 'not_executed', 'research_admission': 'insufficient_evidence',
                   'static_imports': imports}
        print('SYNTHETIC_SUMMARY ' + json.dumps(summary, sort_keys=True))
        return 0 if summary['status'] == 'PASS' else 1
    except Exception as exc:
        print('FAIL: ' + type(exc).__name__ + ': ' + str(exc), file=sys.stderr)
        return 1


if __name__ == '__main__':
    sys.exit(main())
